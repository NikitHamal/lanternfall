cmake_minimum_required(VERSION 3.16)
project(lanternfall LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(NOT MSVC)
  set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG -fno-exceptions -fno-rtti -ffunction-sections -fdata-sections")
else()
  set(CMAKE_CXX_FLAGS_RELEASE "/O2 /DNDEBUG")
endif()
include(FetchContent)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.4 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(glfw)
add_executable(lanternfall src/main.cpp src/game.cpp src/fx.cpp src/audio.cpp src/render.cpp src/meshes.cpp src/ui.cpp src/figure.cpp src/play.cpp src/settings.cpp)
target_link_libraries(lanternfall PRIVATE glfw)
if(UNIX AND NOT APPLE)
  target_link_libraries(lanternfall PRIVATE pthread dl m)
  target_link_options(lanternfall PRIVATE -Wl,--gc-sections -s)
elseif(APPLE)
  target_link_libraries(lanternfall PRIVATE "-framework CoreFoundation" "-framework CoreAudio" "-framework AudioToolbox")
  target_link_options(lanternfall PRIVATE -Wl,-dead_strip -Wl,-x)
elseif(WIN32)
  if(MSVC)
    target_link_options(lanternfall PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:mainCRTStartup)
  else()
    target_link_options(lanternfall PRIVATE -mwindows -s -static)
  endif()
endif()
add_custom_command(TARGET lanternfall POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/assets $<TARGET_FILE_DIR:lanternfall>/assets)
if(WIN32)
  target_compile_definitions(lanternfall PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
endif()
